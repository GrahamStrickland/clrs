% Exercise A.1-5

\begin{proof}
  For $\lvert x \rvert < 1$ we first evaluate the related series in closed form
  by treating it as a geometric series in $x^2$:
  \[
    \sum_{k = 0}^\infty x^{2k + 1} = x \sum_{k = 0}^\infty {(x^2)}^k
    = \frac{x}{1 - x^2}.
  \]
  Since a power series may be differentiated term by term inside its interval of
  convergence, and $\frac{d}{dx} x^{2k + 1} = (2k + 1)x^{2k}$, we have
  \begin{equation*}
    \begin{split}
      \sum_{k = 0}^\infty (2k + 1)x^{2k}
      &= \frac{d}{dx} \sum_{k = 0}^\infty x^{2k + 1} \\
      &= \frac{d}{dx} \frac{x}{1 - x^2} \\
      &= \frac{(1 - x^2) - x \cdot (-2x)}{{(1 - x^2)}^2} \\
      &= \frac{1 - x^2 + 2x^2}{{(1 - x^2)}^2} \\
      &= \frac{1 + x^2}{{(1 - x^2)}^2}.
    \end{split}
  \end{equation*}
\end{proof}
