% Exercise 3.1-6

Let $M(n)$ denote the worst-case running time of the algorithm on inputs of size
$n$ and $m(n)$ its best-case running time, so that for every input of size $n$
with running time $f(n)$ we have $m(n) \leq f(n) \leq M(n)$.

\begin{proof}
  \begin{itemize}
    \item[$(\Rightarrow)$] Suppose the running time of the algorithm is
      $\Theta(g(n))$, i.e., $0 \leq c_1 g(n) \leq f(n) \leq c_2 g(n)$ for all
      $n \geq n_0$ and every input of size $n$, where $c_1, c_2 \in
      \mathbb{R}^+, n, n_0 \in \mathbb{N}$.

      The worst-case time $M(n)$ is the running time of some particular input of
      size $n$, so it satisfies the same bounds; in particular
      $0 \leq M(n) \leq c_2 g(n)$ for all $n \geq n_0$, and $M(n) = O(g(n))$.

      Likewise $m(n)$ is the running time of some particular input of size $n$,
      so $0 \leq c_1 g(n) \leq m(n)$ for all $n \geq n_0$, and
      $m(n) = \Omega(g(n))$.
    \item[$(\Leftarrow)$] Now, suppose the worst-case running time
      $M(n) = O(g(n))$ and the best-case running time $m(n) = \Omega(g(n))$.
      Then we have $0 \leq c_1 g(n) \leq m(n)$ and $0 \leq M(n) \leq c_2 g(n)$
      for all $n \geq n_0$, where $c_1, c_2 \in \mathbb{R}^+, n, n_0 \in
      \mathbb{N}$. But $m(n) \leq f(n) \leq M(n)$ for every input of size $n$,
      so that it follows that
      \[
        0 \leq c_1 g(n) \leq m(n) \leq f(n) \leq M(n) \leq c_2 g(n)
      \]
      for all $n \geq n_0$, i.e., $f(n) = \Theta(g(n))$.
  \end{itemize}
\end{proof}
